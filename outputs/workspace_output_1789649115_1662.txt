#!/usr/bin/env python3
import asyncio, os
from playwright.async_api import async_playwright

BASE = "file://" + os.path.abspath(".")

async def main():
    os.makedirs(".screenshots/logo", exist_ok=True)
    async with async_playwright() as p:
        b = await p.chromium.launch()
        pg = await b.new_page(viewport={"width":1440,"height":900})
        errs=[]
        pg.on("console", lambda m: errs.append(m.text) if m.type=="error" and "jobs.json" not in m.text else None)
        pg.on("pageerror", lambda e: errs.append(str(e)))
        for f in ["index.html","about.html","hire-talent.html"]:
            await pg.goto(f"{BASE}/{f}", wait_until="networkidle")
            await pg.wait_for_timeout(900)
            await pg.screenshot(path=f".screenshots/logo/{f.replace('.html','')}.png", clip={"x":0,"y":0,"width":1440,"height":300})
            txt = await pg.evaluate("() => document.querySelector('.logo-text b').innerText")
            mark = await pg.evaluate("() => !!document.querySelector('.site-header .logo-mark')")
            print(f"{f:20s} logoText='{txt}' hasGoldMark={mark}")
        print("Errors:", len(errs))
        for e in errs[:5]: print("  ",e)
        await b.close()

asyncio.run(main())
