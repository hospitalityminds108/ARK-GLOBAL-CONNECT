  window.ARK_initGenericForm = function (formId, opts) {
    opts = opts || {};
    var form = document.getElementById(formId);
    if (!form) return;
    var cvState = opts.cv ? initFileDrop(opts.cv.drop, opts.cv.input, opts.cv.selected, opts.cv.name, opts.cv.remove, opts.cv.error) : null;
    var btn = form.querySelector('button[type="submit"]');

    form.addEventListener("submit", function (ev) {
      ev.preventDefault();
      var ok = validateForm(form);
      if (opts.cv && (!cvState || !cvState.file)) {
        ok = false;
        var ce = document.getElementById(opts.cv.error);
        if (ce) { ce.textContent = "Please upload your CV."; ce.classList.add("show"); }
      }
      if (!ok) { window.ARK.toast("Please correct the highlighted fields.", "error"); return; }

      var data = {};
      new FormData(form).forEach(function (v, k) { data[k] = v; });
      var subject = (opts.subject || "New Enquiry") + (data.name ? " – " + data.name : "");
      var body = Object.keys(data).map(function (k) { return k + ": " + data[k]; }).join("\n");
      if (cvState && cvState.file) body += "\ncv: " + cvState.file.name + " (" + humanSize(cvState.file.size) + ")";

      btn.disabled = true;
      var original = btn.innerHTML;
      btn.innerHTML = "Submitting…";

      sendEmail({ to_email: CFG.COMPANY_EMAIL, subject: subject, message: body, name: data.name || "", email: data.email || "" })
        .then(function (res) {
          btn.disabled = false;
          btn.innerHTML = original;
          var box = document.getElementById(opts.successId || "formSuccess");
          var note = res.ok ? "Your message has been sent to the ARK team."
            : (res.demo ? "Email is in demo mode (EmailJS not configured). Your details were captured locally — please also send them on WhatsApp."
              : "We could not reach the email service. Please send your details on WhatsApp.");
          if (box) {
            box.classList.remove("hidden");
            box.innerHTML = '<div class="alert alert-success">' + ICONS.checkCircle + "<span>" + note + "</span></div>" +
              '<div class="mt-2"><a class="btn btn-gold" target="_blank" rel="noopener" href="' + waUrl(subject + "\n\n" + body) + '">' + ICONS.whatsapp + " Send on WhatsApp</a></div>";
            box.scrollIntoView({ behavior: "smooth", block: "center" });
          }
          form.reset();
          if (cvState) { cvState.file = null; var s = document.getElementById(opts.cv.selected); if (s) s.classList.remove("show"); var d = document.getElementById(opts.cv.drop); if (d) d.style.display = ""; }
          window.ARK.toast("Submitted successfully.", "success");
        });
    });
  };

  /* ---------- Live validation clearing ---------- */
  document.addEventListener("input", function (ev) {
    if (ev.target.matches("input, select, textarea")) clearError(ev.target);
  });
})();
